cmake_minimum_required(VERSION 3.8)
project(embodied_comm_cpp)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(rcl_interfaces REQUIRED)
find_package(std_msgs REQUIRED)
find_package(jsoncpp REQUIRED)

foreach(node sensor_simulator status_monitor)
  add_executable(${node} src/${node}.cpp)
  target_include_directories(${node} PRIVATE include)
  target_compile_options(${node} PRIVATE -Wall -Wextra -Wpedantic)
  ament_target_dependencies(${node} rclcpp rcl_interfaces std_msgs)
  target_link_libraries(${node} JsonCpp::JsonCpp)
  install(TARGETS ${node} DESTINATION lib/${PROJECT_NAME})
endforeach()
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  ament_add_gtest(test_protocol test/test_protocol.cpp)
  target_include_directories(test_protocol PRIVATE include)
  target_link_libraries(test_protocol JsonCpp::JsonCpp)
endif()
ament_package()
